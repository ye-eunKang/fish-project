let lastTiltAngle = null; // 가로 모드 기준 센서 감지 함수 function handleOrientation(e) { if (isDropping || isFinished) return; // 현재 기기의 회전 방향 확인 (90: 시계방향 가로, -90 또는 270: 반시계방향 가로) const orientationAngle = (screen.orientation && screen.orientation.angle !== undefined) ? screen.orientation.angle : (window.orientation || 0); const beta = e.beta || 0; // 세로 기준 앞뒤 기울기 (-180 ~ 180) const gamma = e.gamma || 0; // 세로 기준 좌우 기울기 (-90 ~ 90) // 가로 방향에 따른 실시간 좌우 틸트 각도 계산 let currentTilt = 0; if (Math.abs(orientationAngle) === 90 || orientationAngle === 270) { // 가로 모드일 때는 beta가 사용자의 체감상 '좌우 틸트'가 됨 // 폰 머리가 왼쪽인지 오른쪽인지에 따라 방향 부호 반전 보정 currentTilt = (orientationAngle === 90) ? beta : -beta; } else { // 혹시 가로 고정이 안 풀렸거나 데스크톱/기타 환경일 때의 폴백 (gamma 사용) currentTilt = gamma; } // 1. 민감도 설정: 가로 화면은 폭(width)이 넓으므로 민감도 계수를 약 18~22로 세팅 // 살짝(±10도 내외)만 양손을 기울여도 집게가 가로 화면 전체를 부드럽게 커버함 const SENSITIVITY = 20; const screenW = window.innerWidth; targetX = (screenW / 2) + (currentTilt * SENSITIVITY); // 화면 양끝 이탈 방지 마진 (좌우 60px) targetX = Math.max(60, Math.min(screenW - 60, targetX)); // 2. 가만히 있는 상태(정지 범위) 판정: // 가로 모드로 양손으로 쥐면 한 손 세로 파지보다 안정적이지만, // 여전히 관객의 미세한 맥박/호흡 떨림을 감안해 1.5도 이내의 흔들림은 '정지'로 간주 (민감도 낮춤) if (lastTiltAngle !== null) { const delta = Math.abs(currentTilt - lastTiltAngle); if (delta < 1.5) { if (!stillStartTime) stillStartTime = performance.now(); } else { // 손을 의도적으로 기울여 이동시키면 프로그레스 리셋 stillStartTime = null; updateProgress(0); } } lastTiltAngle = currentTilt; }
